def latex_escape(text: str) -> str:
    replacements = {
        "\\": r"\textbackslash{}",
        "&": r"\&",
        "%": r"\%",
        "$": r"\$",
        "#": r"\#",
        "_": r"\_",
        "{": r"\{",
        "}": r"\}",
        "~": r"\textasciitilde{}",
        "^": r"\textasciicircum{}",
    }
    return "".join(replacements.get(char, char) for char in text)


def latex_path(text: str) -> str:
    return latex_escape(text.replace("\\", "/"))


def read_text(path: Path) -> str:
    for encoding in ("utf-8-sig", "utf-8", "cp1252", "latin-1"):
        try:
            return path.read_text(encoding=encoding)
        except UnicodeDecodeError:
            continue
    raise UnicodeDecodeError("unknown", b"", 0, 1, "Could not decode input file")


def parse_header_value(text: str, label: str) -> str:
    match = re.search(rf"^{re.escape(label)}\s*:\s*(.+)$", text, flags=re.MULTILINE)
    return match.group(1).strip() if match else ""


def parse_entries(text: str) -> list[dict[str, str]]:
    separator = "=" * 78
    parts = text.split(separator)
    entries: list[dict[str, str]] = []

    for part in parts[1:]:
        part = part.strip()
        if not part or "Source file:" not in part:
            continue

        metadata, divider, body = part.partition("-" * 78)
        if not divider:
            continue

        fields: dict[str, str] = {}
        field_map = {
            "Source file": "file",
            "Folder": "folder",
            "Source path": "path",
            "Date modified": "modified",
            "Paragraph": "paragraph",
            "Excerpt type": "type",
            "Matched term(s)": "terms",
            "Note": "note",
        }

        for line in metadata.splitlines():
            if ":" not in line:
                continue
            key, value = line.split(":", 1)
            key = key.strip()
            if key in field_map:
                fields[field_map[key]] = value.strip()

        body = body.strip()
        if fields.get("file") and body:
            fields["body"] = body
            entries.append(fields)

    return entries


def highlight_terms(text: str, terms: list[str]) -> str:
    escaped_terms = [latex_escape(term) for term in terms if term.strip()]
    escaped_text = latex_escape(text)

    for term in sorted(escaped_terms, key=len, reverse=True):
        if not term:
            continue
        pattern = re.compile(re.escape(term), flags=re.IGNORECASE)
        escaped_text = pattern.sub(lambda match: r"\hit{" + match.group(0) + "}", escaped_text)

    return escaped_text


def render_entry(entry: dict[str, str], terms: list[str]) -> str:
    key_order = ["file", "folder", "path", "modified", "paragraph", "type", "terms", "note"]
    key_names = {
        "file": "file",
        "folder": "folder",
        "path": "path",
        "modified": "modified",
        "paragraph": "paragraph",
        "type": "type",
        "terms": "terms",
        "note": "note",
    }

    lines = [r"\begin{excerpt}{"]
    for key in key_order:
        value = entry.get(key, "").strip()
        if not value:
            continue
        safe_value = latex_path(value) if key == "path" else latex_escape(value)
        lines.append(f"  {key_names[key]:<10}= {{{safe_value}}},")
    lines.append("}")

    body_paragraphs = re.split(r"\n\s*\n", entry["body"])
    lines.append("\n\n".join(highlight_terms(paragraph.strip(), terms) for paragraph in body_paragraphs if paragraph.strip()))
    lines.append(r"\end{excerpt}")
    return "\n".join(lines)
