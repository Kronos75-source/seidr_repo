def read_text_file(file_path: Path) -> str:
    """Read common Windows text encodings without altering the source."""
    for encoding in ("utf-8", "utf-8-sig", "cp1252", "latin-1"):
        try:
            return file_path.read_text(encoding=encoding)
        except UnicodeDecodeError:
            continue

    raise UnicodeError(
        f"Could not decode source file:\n{file_path}"
    )


def parse_source_filename(file_path: Path) -> tuple[str, str]:
    """
    Parse:
        2026-08-28 — With Rán.txt
    into:
        ('2026-08-28', 'With Rán').

    If no YYYY-MM-DD prefix is available, use the file's modified date.
    """
    stem = file_path.stem.strip()

    match = re.match(
        r"^(?P<date>\d{4}-\d{2}-\d{2})"
        r"\s*(?:—|–|-)\s*"
        r"(?P<title>.+?)\s*$",
        stem,
    )

    if match:
        return match.group("date"), match.group("title").strip()

    fallback_date = datetime.fromtimestamp(
        file_path.stat().st_mtime
    ).strftime("%Y-%m-%d")

    return fallback_date, stem


def make_record_slug(text: str) -> str:
    """
    Produce an ASCII file-safe record-number portion.

    Examples:
        With Rán -> WITH-RAN
        Skaði Notes -> SKADI-NOTES
        Offering & Dreamwork -> OFFERING-DREAMWORK
    """
    normalized = unicodedata.normalize("NFKD", text)

    normalized = "".join(
        character
        for character in normalized
        if not unicodedata.combining(character)
    )

    normalized = normalized.upper()
    normalized = re.sub(r"[^A-Z0-9]+", "-", normalized)

    return normalized.strip("-") or "UNTITLED"


def safe_folder_name(text: str) -> str:
    """Return a Windows-safe version of the repository division name."""
    invalid_characters = r'<>:"/\|?*'

    cleaned = text.strip()

    for character in invalid_characters:
        cleaned = cleaned.replace(character, "-")

    cleaned = re.sub(r"\s+", " ", cleaned)
    cleaned = cleaned.strip(" .")

    return cleaned or "Uncategorized"


def latex_escape(text: str) -> str:
    """
    Escape LaTeX syntax characters but preserve ordinary Unicode characters.
    """
    replacements = {
        "\\": r"\textbackslash{}",
        "&": r"\&",
        "%": r"\%",
        "$": r"\$",
        "#": r"\#",
        "_": r"\_",
        "{": r"\{",
        "}": r"\}",
        "~": r"\textasciitilde{}",
        "^": r"\textasciicircum{}",
    }

    return "".join(
        replacements.get(character, character)
        for character in text
    )


def render_source_entry(text: str) -> str:
    """
    Preserve blank-line paragraph breaks from the source .txt note.
    """
    normalized = text.replace("\r\n", "\n").replace("\r", "\n")
    raw_paragraphs = re.split(r"\n\s*\n+", normalized)

    rendered_paragraphs = []

    for paragraph in raw_paragraphs:
        clean_lines = [
            line.strip()
            for line in paragraph.split("\n")
            if line.strip()
        ]

        if clean_lines:
            rendered_paragraphs.append(
                latex_escape(" ".join(clean_lines))
            )

    if not rendered_paragraphs:
        return r"\noindent\textit{No readable source text was found.}"

    return "\n\n".join(rendered_paragraphs)
