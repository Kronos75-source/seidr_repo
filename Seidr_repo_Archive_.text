def read_text_file(file_path: Path) -> str:
    """Read the source note without altering it."""
    for encoding in ("utf-8", "utf-8-sig", "cp1252", "latin-1"):
        try:
            return file_path.read_text(encoding=encoding)
        except UnicodeDecodeError:
            continue

    raise UnicodeError(
        f"Could not decode source file:\n{file_path}"
    )


def parse_source_filename(file_path: Path) -> tuple[str, str]:
    """
    Parse:
        2026-06-01 — Doen.txt
    into:
        ('2026-06-01', 'Doen')
    """
    stem = file_path.stem.strip()

    match = re.match(
        r"^(?P<date>\d{4}-\d{2}-\d{2})"
        r"\s*(?:—|–|-)\s*"
        r"(?P<title>.+?)\s*$",
        stem,
    )

    if match:
        return match.group("date"), match.group("title").strip()

    fallback_date = datetime.fromtimestamp(
        file_path.stat().st_mtime
    ).strftime("%Y-%m-%d")

    return fallback_date, stem


def make_record_slug(text: str) -> str:
    """Create a stable filename-safe record identifier segment."""
    normalized = unicodedata.normalize("NFKD", text)

    normalized = "".join(
        char
        for char in normalized
        if not unicodedata.combining(char)
    )

    normalized = normalized.upper()
    normalized = re.sub(r"[^A-Z0-9]+", "-", normalized)

    return normalized.strip("-") or "UNTITLED"


def infer_record_type(folder_name: str) -> str:
    """Assign an editable initial classification from the folder name."""
    mapping = {
        "Body": "Somatic Record",
        "David Advice": "Correspondence Record",
        "Freyja": "Entity-Focused Record",
        "Lilith": "Entity-Focused Record",
        "Skaði": "Entity-Focused Record",
    }

    return mapping.get(folder_name, "Journal Record")


# -------------------------------------------------------------------
# LATEX-SAFE PROSE
# -------------------------------------------------------------------

def latex_escape(text: str) -> str:
    """
    Escape characters that would otherwise have special meanings in LaTeX.
    """
    replacements = {
        "\\": r"\textbackslash{}",
        "&": r"\&",
        "%": r"\%",
        "$": r"\$",
        "#": r"\#",
        "_": r"\_",
        "{": r"\{",
        "}": r"\}",
        "~": r"\textasciitilde{}",
        "^": r"\textasciicircum{}",
    }

    return "".join(
        replacements.get(character, character)
        for character in text
    )


def render_source_entry(text: str) -> str:
    """
    Convert blank-line-separated .txt paragraphs into normal LaTeX prose.
    """
    normalized = text.replace("\r\n", "\n").replace("\r", "\n")
    raw_paragraphs = re.split(r"\n\s*\n+", normalized)

    output_paragraphs = []

    for paragraph in raw_paragraphs:
        lines = [
            line.strip()
            for line in paragraph.split("\n")
            if line.strip()
        ]

        if lines:
            output_paragraphs.append(
                latex_escape(" ".join(lines))
            )

    if not output_paragraphs:
        return r"\noindent\textit{No readable source text was found.}"

    return "\n\n".join(output_paragraphs)
